# Preserve the previous TDD granularity: one executable and CTest entry per source module.
function(add_tiny_unit_test target source ctest_name toolkit_target)
    add_executable(${target} "${source}")
    target_link_libraries(${target} PRIVATE ${toolkit_target} jh::test::jh-tiny-test)
    target_compile_features(${target} PRIVATE cxx_std_20)
    add_test(NAME "${ctest_name}" COMMAND ${target})
endfunction()
